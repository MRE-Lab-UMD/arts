# hardware, not the arm it sits on. "psm1"/"psm2" in the comments below are the designer's names.
# FORCEPS: tool-specific homing values (forceps_01 calibration, 2-4 Nov 2026).
# Loaded by tool_homing with tool:=forceps (the GUI's "Forceps" button passes this).
# Every key below is REQUIRED: a missing or unknown key stops the node before any motor moves, so a new
# tool file (scissors.txt, ...) must set every value on purpose instead of silently inheriting forceps values.
# Format: <key> <value>   ('#' starts a comment). Ticks: 4096 per motor turn. Currents: raw Dynamixel units.
# Edits take effect the next time the node starts (no rebuild needed).
# A ROS parameter given on the command line (e.g. -p jaw.max_close:=1500) still overrides a value here.

homing.jaw_mode contact          # jaw sweep + close-onto-contact plan ("limits" = one jaw with hard stops, like the hook)

# ---------------- jaws: motion ----------------
jaw.sweep_current 75             # goal current during jaw sweeps / closes
jaw.sweep_profile_velocity 22

jaw.move_fast_profile 100        # M5 back to the middle, fast part ...
jaw.move_fast_current 150        # ... with more current
jaw.move_slow_margin 200         # gentle for the last N ticks before the target ...

jaw.jaw_final_hold 75

# ---------------- jaws: finding the center ----------------
# Forceps: M5 is placed from its load step (or its engaged start); M2 then closes until it TOUCHES M5.
jaw.m5_step_offset 107           # default step - center if the adapter has no entry below
jaw.m2_step_offset 139
jaw.m2_contact_offset -4         # M2 contact position - center (M5 centered)
jaw.m2_apply_contact_offset false # false: M2 parks AT the detected contact (offset only cross-checks it)
jaw.m2_center_snap_max 300       # (only with <adapter>.m2_center_angle) M2 snaps after the contact only within this
jaw.m5_center_snap_max 300       # (only with <adapter>.m5_center_angle) step mode snaps M5 only within this
jaw.m5_engaged_start 63          # engaged M5 sweep start - center
jaw.engaged_trend 1.10           # M5 "engaged at start": current climb 450-600 vs 150-300 ticks
jaw.m2_engaged_trend 1.06
jaw.m5_engaged_mid 1.025         # ... and already climbing at 300-450 (else a step starting: keep sweeping)
jaw.m5_defer_travel 1300
jaw.stop_after_step 450
jaw.check_tol 80                 # M2 step vs contact disagreement that raises a JAW FLAG
jaw.wall_current 150             # held jaw's current while the other closes onto it
jaw.collapse_ratio 0.8           # contact = at current limit AND speed < ratio x cruising
jaw.goto_push_ratio 0.15         # M5 move back: nearly stalled at the limit = blocked
